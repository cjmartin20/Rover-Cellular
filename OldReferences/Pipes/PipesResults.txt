//in file anonymous_pipes.c
#include <stdio.h>
#include <unistd.h>
#include <time.h>
#include <wait.h>
#include <string.h>
#include "randapi.h"

#define MaxLen 80
#define PipeStdIn 0 //stdin for Unix is 0
#define PipeStdOut 1 //stdout for Unix is 1

int main ( ) { //bridge 

	int ret, ret2, comPipes[2]; //comPipes[0] for input, comPipes[1] for output
	char buff[ MaxLen + 1 ];

	time_t t = time(NULL);
	struct tm startTime = *localtime( &t );

	printf( "---------------------------------\n" );
	printf( "USS Enterprise Systems Monitoring\n" );
	printf( "TIME ON DECK [%d:%d:%d]\n", startTime.tm_hour, startTime.tm_min, startTime.tm_sec);
	printf( "---------------------------------\n" );

	if( pipe( comPipes ) == 0 ) { //create pipes
	// seed random number generator
	seedRandGenerator( );

		pid_t navID = fork( );
		if ( navID == 0) { // create navigation, child 1
			
			pid_t lsID = fork( );
			if( lsID == 0 ) { // create life support – child 2
				struct tm currTime = *localtime( &t );
				float lsTime;

				char lsBuff[ MaxLen + 1 ];
				char *gasFormat = {"LS  : breathing gas levels adjustment	 	- Elapsed Time : %5.3f sec"};
				char *litFormat = {"LS  : temperature and lighting adjustment	- Elapsed Time : %5.3f sec"};
				int n = 1; 

				while( n > 0 ) {
					//adjust gas levels
					lsTime = getRandExponential() * 5.0;
					sleep( lsTime );
					sprintf(&lsBuff[0], gasFormat, lsTime);
					ret = write( comPipes[ PipeStdOut ], lsBuff, MaxLen );
					//adjust lighting and temperature
					lsTime = getRandExponential() * 4.0;
					sleep( lsTime );
					sprintf(&lsBuff[0], litFormat, lsTime);
					ret = write( comPipes[ PipeStdOut ], lsBuff, MaxLen );
					ret = wait( NULL );
					t = time(NULL);
					currTime = *localtime( &t );
					n--;
				} // end while - life support loop
			} //end life support process
			else {
				// child1 navigation
				struct tm currTime = *localtime( &t );
				float navTime;

				char navBuff[ MaxLen + 1 ];
				char *navFormat = {"NAV : navigation bearing adjustment		- Elapsed Time : %5.3f sec"};
				int n = 1; 

				while( n > 0 ) {
					navTime = getRandFloat() * 6.0;
					sleep( navTime );
					sprintf(&navBuff[0], navFormat, navTime);
					ret = write( comPipes[ PipeStdOut ], navBuff, MaxLen );
					t = time(NULL);
					currTime = *localtime( &t );
					n--;
				}  // end while – navigation loop
			} //end navigation process
			// code common to life support and navigation
			ret = wait( NULL );
		}
		else  { // bridge
			struct tm currTime = *localtime( &t );
			int knt = 3;
			while ( knt > 0 ) { //for later use startTime.tm_min + 1 > currTime.tm_min
				ret = read( comPipes[ PipeStdIn ], buff, MaxLen );
				printf( "%s", buff );
				t = time(NULL);
				currTime = *localtime( &t );
				printf( " RECIEVED AT BRIDGE [%02d:%02d:%02d]\n", currTime.tm_hour, currTime.tm_min, currTime.tm_sec);
				knt--;
			}  //end while
		}  // end bridge specific
		// code common to bridge and life support
		close( comPipes[ PipeStdIn ] );
		close( comPipes[ PipeStdOut ] );
	}
	return 0;
} // end program

/*
 

	Random Functions API Wrapper in file randapi.c

	extern float getRandFloat(  );   returns random floating point 0 <= rand < 1.0.
	extern getRandInteger(  max );    returns random integer between 0 and (max - 1)
	extern float getRandExponential(  ); returns random exponential
	      inter-arrival time, multiply by the mean to obtain the actual interval.

	Must seed first using extern void seedGenerator( void );
	in file seedRandGererator.c.
	Prototypes in randapi.h

	All “C” compilers provide a random number generator providing random integers
	In the range 0 =< rand( ) < RAND_MAX where RAND_MAX is compiler/hardware
	 dependent.  The quality of the deviates is a topic of constant debate.  “rand( )”
	is uniformly distributed.  This functionality is provided in stdlib.h.  srand(  ) may be
	used to seed the generator.  If the seed is not changed/set, rand(  ) will always
	produce the same series (sequence)of random numbers - a pseudo random
	 number generator.
*/

/* generate random floating point */
#include <stdlib.h>

float getRandFloat( )    //  0 <= random deviate < 1
{					// “RAND_MAX” is largest integer
		float randDeviate;

		randDeviate = ((float)rand( ) / (float)RAND_MAX);
		return randDeviate;

		//return ((float)rand( ) / (float)RAND_MAX);
}

/* generate random integer from 0 thru (max - 1).  */
int getRandInteger( int max )
{
	int randDeviate;
	randDeviate = (int)((float)max * rand( ) / (RAND_MAX+1.0));  //force float division
	return randDeviate;
}

/*
	Initialize Random API using system clock.
	This is convenient but not cryptographically sound.
*/

#include <stdlib.h>
#include <time.h>

void seedRandGenerator( )
{
	time_t seed;  //The seed repeats every 60 seconds.

	seed = time( NULL );    // srand( time( NULL ) );
	srand( seed );
	return;
}
